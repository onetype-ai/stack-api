==> src/utils/Order.ts

class OrderUtils
{
    forLocale(locale: string): Intl.Collator
    {
        try
        {
            return new Intl.Collator(locale, { sensitivity: "variant" });
        }
        catch
        {
            return new Intl.Collator("en", { sensitivity: "variant" });
        }
    }
}

export const Order = new OrderUtils();

==> src/utils/Text.ts

class TextUtils
{
    static #strokes = new Map(Object.entries({
        ø: "o", đ: "d", ł: "l", ħ: "h", ŧ: "t", ƀ: "b", ɖ: "d", ƶ: "z", ə: "e",
        æ: "ae", œ: "oe", ß: "ss", þ: "th", ð: "d", ŋ: "n", ı: "i",
    }));

    static #ligatures = new Map(Object.entries({
        "\uFB00": "ff", "\uFB01": "fi", "\uFB02": "fl",
        "\uFB03": "ffi", "\uFB04": "ffl", "\uFB05": "st", "\uFB06": "st",
    }));

    static #joiners: ReadonlySet<string> = new Set(["\u200C", "\u200D"]);

    searchable(raw: string): string
    {
        return this.#plain(raw)
            .replace(/\p{Diacritic}/gu, "")
            .replace(/[øđłħŧƀɖƶəæœßþðŋı]/gu, (letter) => TextUtils.#strokes.get(letter) ?? letter);
    }

    comparable(raw: string): string
    {
        return this.#plain(raw)
            .replace(/[\uFB00-\uFB06]/gu, (ligature) => TextUtils.#ligatures.get(ligature) ?? ligature)
            .normalize("NFC");
    }

    characters(raw: string): number
    {
        return [...new Intl.Segmenter().segment(raw)].length;
    }

    visible(raw: string): string
    {
        return raw.replace(/[\p{Cf}\u00AD\u200B\uFEFF]/gu, (mark) =>
        {
            return TextUtils.#joiners.has(mark) ? mark : "";
        });
    }

    #plain(raw: string): string
    {
        return this.visible(raw)
            .normalize("NFD")
            .toLowerCase()
            .trim()
            .replace(/\s+/gu, " ");
    }
}

export const Text = new TextUtils();
