==> src/plugins/notes/usage.md

# notes

## Description

Notes a reader keeps: written, listed by cursor, searched by title in any
language, and seen by nobody else.

## Purpose

The plugin every mechanic is shown on. It owns its own tables, narrows every
read to one reader, opens a point another plugin may refuse at, and announces
what happened without knowing who hears.

## Usage

```ts
import { Notes } from "@plugins/notes";

const page = await Notes.list(ctx);
const note = await Notes.get(ctx, id);
const many = await Notes.many(ctx);
```

Routes: `GET /notes`, `GET /notes/search`, `GET /notes/:id`, `POST /notes`,
`DELETE /notes/:id`. Each carries a budget.

- A page is a cursor, never an offset: `after` names the sequence the last row
  carried, and the query does the limiting.
- Sequence is taken inside the write, not from the clock, because two notes
  written in one millisecond would tie.
- Matching is a folded column; ordering is `Intl.Collator`. SQLite folds A to
  Z and sorts by code point, so neither works in SQL.
- `notes.note.before-writing` lets another plugin refuse with a reason.

## Refuses

- A title outside 1 to 120 characters, counted as a reader sees them.
- A second note by a title already taken, however it was typed.
- More than `mostPerOwner`, counted inside the transaction that writes.
- Somebody else's note, as 404 rather than 403.
- A reader carrying no claim, rather than falling back to a default.

import { Refusal } from "@onetype/stack-api-kit";
import { z } from "zod";

import { Text } from "@utils/Text";

const MostCharacters = 120;

export const Note = {
    schema: z.object({
        id: z.uuid(),
        title: z.string(),
        body: z.string(),
        writtenAt: z.string(),
    }),

    parseTitle: (raw: string): string =>
    {
        const title = Text.visible(raw).normalize("NFC").trim();

        if (Text.characters(title) < 1 || Text.characters(title) > MostCharacters)
        {
            throw new Refusal(400, "BAD_TITLE", `A title is 1 to ${String(MostCharacters)} characters.`, {
                title: `Between 1 and ${String(MostCharacters)} characters.`,
            });
        }

        return title;
    },
};

export type Note = z.infer<typeof Note.schema>;

import { Refusal } from "@onetype/stack-api-kit";
import { and, asc, count, eq, gt, like, max } from "drizzle-orm";

import { Note } from "../schemas/Note";
import { Order } from "@utils/Order";
import { Text } from "@utils/Text";
import { notes } from "../tables/notes";

import type { NoteQuery } from "../types/NoteQuery";
import type { NotePage } from "../schemas/NotePage";
import type { NotesContext } from "../types/Context";
import type { SQL } from "drizzle-orm";

type Row = typeof notes.$inferSelect;

export class Notes
{
    readonly #ctx: NotesContext;

    constructor(ctx: NotesContext)
    {
        this.#ctx = ctx;
    }

    async list(query: NoteQuery = {}): Promise<NotePage>
    {
        const size = this.#ctx.config.pageSize;

        const rows = await this.#ctx.db
            .select()
            .from(notes)
            .where(this.#whereAfter(query.after))
            .orderBy(asc(notes.sequence))
            .limit(size + 1);

        const page = rows.slice(0, size);
        const last = page.at(-1);

        return {
            notes: page.map((row) => this.#toNote(row)),
            ...(rows.length > size && last !== undefined && { after: last.sequence }),
        };
    }

    async get(id: string): Promise<Note>
    {
        const [row] = await this.#ctx.db.select().from(notes).where(this.#whereId(id));

        if (row === undefined)
        {
            throw this.#notFoundError();
        }

        return this.#toNote(row);
    }

    async search(text: string): Promise<Note[]>
    {
        const needle = Text.searchable(text);

        if (needle === "")
        {
            return [];
        }

        const rows = await this.#ctx.db
            .select()
            .from(notes)
            .where(and(this.#whereOwner(), like(notes.searchable, `%${needle}%`)))
            .limit(this.#ctx.config.pageSize);

        const collator = Order.forLocale(this.#ctx.config.locale);

        return rows
            .sort((first, second) => collator.compare(first.title, second.title))
            .map((row) => this.#toNote(row));
    }

    async write(rawTitle: string, body: string): Promise<Note>
    {
        const title = Note.parseTitle(rawTitle);
        const searchable = Text.searchable(title);
        const comparable = Text.comparable(title);

        const refusal = await this.#ctx.hooks.run("notes.note.before-writing", { title, body });

        if (refusal !== undefined)
        {
            throw new Refusal(409, "NOTE_REFUSED", refusal);
        }

        return this.#ctx.tx(async (inside) =>
        {
            const [tally] = await inside.db
                .select({ many: count() })
                .from(notes)
                .where(this.#whereOwner());

            if ((tally?.many ?? 0) >= this.#ctx.config.mostPerOwner)
            {
                throw new Refusal(409, "TOO_MANY", "That is as many notes as one reader may keep.", {
                    most: String(this.#ctx.config.mostPerOwner),
                });
            }

            const [highest] = await inside.db
                .select({ at: max(notes.sequence) })
                .from(notes)
                .where(this.#whereOwner());

            const row = {
                id: crypto.randomUUID(),
                title,
                searchable,
                comparable,
                body,
                sequence: (highest?.at ?? 0) + 1,
                writtenAt: new Date(this.#ctx.now()).toISOString(),

                ...this.#ctx.stamped("notes"),
            } as Row;

            await inside.db.insert(notes).values(row).onConflictDoNothing();

            const [saved] = await inside.db.select().from(notes).where(this.#whereId(row.id));

            if (saved === undefined)
            {
                throw new Refusal(409, "TITLE_TAKEN", "A note by that title is already here.");
            }

            inside.events.emit("notes.note.written", { id: row.id, ownerId: row.ownerId });

            return this.#toNote(saved);
        });
    }

    async remove(id: string): Promise<void>
    {
        await this.#ctx.tx(async (inside) =>
        {
            const rows = await inside.db
                .delete(notes)
                .where(this.#whereId(id))
                .returning({ id: notes.id, ownerId: notes.ownerId });

            const row = rows[0];

            if (row === undefined)
            {
                throw this.#notFoundError();
            }

            inside.events.emit("notes.note.removed", { id, ownerId: row.ownerId });
        });
    }

    async many(): Promise<number>
    {
        const [tally] = await this.#ctx.db.select({ many: count() }).from(notes).where(this.#whereOwner());

        return tally?.many ?? 0;
    }

    #whereOwner(): SQL | undefined
    {
        return this.#ctx.scoped<SQL>("notes");
    }

    #whereAfter(after?: number): SQL | undefined
    {
        const owner = this.#whereOwner();

        return after === undefined ? owner : and(owner, gt(notes.sequence, after));
    }

    #whereId(id: string): SQL | undefined
    {
        return and(eq(notes.id, id), this.#whereOwner());
    }

    #notFoundError(): Refusal
    {
        return new Refusal(404, "NOT_FOUND", "No such note.");
    }

    #toNote(row: Row): Note
    {
        return { id: row.id, title: row.title, body: row.body, writtenAt: row.writtenAt };
    }
}

==> src/plugins/labels/usage.md

# labels

## Description

Words put on a note, so a reader finds it by more than its title.

## Purpose

The plugin every crossing is shown on. It calls another's public API, hears
what that one announced, joins a point it opened, and answers a command run
for nobody.

## Usage

```ts
import { Labels } from "@plugins/labels";

const words = await Labels.forNote(ctx, noteId);
```

Routes: `GET /notes/:noteId/labels`, `POST /notes/:noteId/labels`.

- Calls `Notes.get` before writing, because the answer decides what happens
  next and an event answers nothing. That is why `notes` is in `dependsOn`.
- Hears `notes.note.removed` and drops what it held. Hearing is not depending:
  the listener names nothing.
- Joins `notes.note.before-writing`, refusing a title carrying a refused word.
- `labels.forget-note` runs for nobody, so whose work it is travels in the
  input.

## Refuses

- A label on a note that is missing or somebody else's, both as 404.
- A word in `refused`, or one that folds away to nothing.
- More than `mostPerNote`, counted inside the transaction.

import { Refusal } from "@onetype/stack-api-kit";
import { and, count, eq } from "drizzle-orm";

import { Notes } from "@plugins/notes";
import { Text } from "@utils/Text";
import { labels } from "../tables/labels";

import type { Label } from "../schemas/Label";
import type { LabelsContext } from "../types/Context";
import type { SQL } from "drizzle-orm";

type Row = typeof labels.$inferSelect;

export class Labels
{
    readonly #ctx: LabelsContext;

    constructor(ctx: LabelsContext)
    {
        this.#ctx = ctx;
    }

    async forNote(noteId: string): Promise<Label[]>
    {
        const rows = await this.#ctx.db
            .select()
            .from(labels)
            .where(this.#whereNote(noteId))
            .limit(this.#ctx.config.mostPerNote);

        return rows.map((row) => this.#toLabel(row));
    }

    async add(noteId: string, rawWord: string): Promise<Label>
    {
        const put = await this.#write(noteId, rawWord);

        this.#ctx.push("labels.put", { noteId: put.noteId, word: put.word });

        return put;
    }

    async #write(noteId: string, rawWord: string): Promise<Label>
    {
        const word = Text.comparable(rawWord);

        if (word === "")
        {
            throw new Refusal(400, "EMPTY_WORD", "A label is at least one character.");
        }

        if (this.#ctx.config.refused.includes(word))
        {
            throw new Refusal(409, "WORD_REFUSED", `"${word}" is not a label anyone may use.`);
        }

        await Notes.get.one(this.#ctx, noteId);

        return this.#ctx.tx(async (inside) =>
        {
            const [tally] = await inside.db
                .select({ many: count() })
                .from(labels)
                .where(this.#whereNote(noteId));

            if ((tally?.many ?? 0) >= this.#ctx.config.mostPerNote)
            {
                throw new Refusal(409, "TOO_MANY", "That note carries as many labels as one may.", {
                    most: String(this.#ctx.config.mostPerNote),
                });
            }

            const row = {
                id: crypto.randomUUID(),
                noteId,
                word,
                ...this.#ctx.stamped("labels"),
            } as Row;

            await inside.db.insert(labels).values(row).onConflictDoNothing();

            const [saved] = await inside.db
                .select()
                .from(labels)
                .where(and(this.#whereNote(noteId), eq(labels.word, word)));

            if (saved === undefined)
            {
                throw new Refusal(500, "NOT_WRITTEN", "The label could not be written.");
            }

            return this.#toLabel(saved);
        });
    }

    async dropFor(noteId: string): Promise<number>
    {
        const rows = await this.#ctx.write(() =>
        {
            return this.#ctx.db
                .delete(labels)
                .where(this.#whereNote(noteId))
                .returning({ id: labels.id });
        });

        return rows.length;
    }

    #whereOwner(): SQL | undefined
    {
        return this.#ctx.scoped<SQL>("labels");
    }

    #whereNote(noteId: string): SQL | undefined
    {
        return and(eq(labels.noteId, noteId), this.#whereOwner());
    }

    #toLabel(row: Row): Label
    {
        return { id: row.id, noteId: row.noteId, word: row.word };
    }
}

    },

    {

    },

==> src/plugins/readers/usage.md

# readers

## Description

Who is calling, and what being them means.

## Purpose

The plugin the composition root does not have to know about. It reads the
request it understands and says who sent it; the kernel fills permissions from
what it grants.

## Usage

```ts
import { Readers } from "@plugins/readers";

const reader = await Readers.get.byKey(ctx, key);
```

- `identifies` reads `x-key` and answers an identity, or nothing. At most one
  plugin declares it, and `main.ts` wires none.
- `grants` answers what a role may do. `identifies` never names a permission,
  so no plugin grants itself one.
- `mayGrant` is every permission any role carries. A route requiring one
  outside it refuses the boot, naming both.
- A key is kept as its SHA-256 fingerprint, never as itself.

## Refuses

Nothing. A key nobody holds is a stranger rather than a refusal: only a closed
route turns that into a 401, and a public one answers a stranger like anyone
else.

==> src/plugins/readers/services/directory.ts

import { eq } from "drizzle-orm";

import { Role } from "../schemas/Role";
import { readers } from "../tables/readers";

import type { ReadersContext } from "../types/Context";

export type Reader = {
    id: string;
    role: Role;
};

export class Directory
{
    readonly #ctx: ReadersContext;

    constructor(ctx: ReadersContext)
    {
        this.#ctx = ctx;
    }

    async of(key: string | undefined): Promise<Reader | undefined>
    {
        if (key === undefined || key === "")
        {
            return undefined;
        }

        const [row] = await this.#ctx.db
            .select()
            .from(readers)
            .where(eq(readers.keyDigest, await Directory.digest(key)));

        if (row === undefined)
        {
            return undefined;
        }

        const role = Role.schema.safeParse(row.role);

        return role.success ? { id: row.id, role: role.data } : undefined;
    }

    grants(role: Role): readonly string[]
    {
        return Role.grants[role];
    }

    static async digest(key: string): Promise<string>
    {
        const bytes = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(key));

        return [...new Uint8Array(bytes)].map((byte) => byte.toString(16).padStart(2, "0")).join("");
    }
}
