# index.ts

The public API: the only file another plugin may import, and only when it
names this one in `dependsOn`. Everything else in the folder is private.

`<Plugin>` is the plugin's folder capitalised; `<service>` is the service it
reaches, which may be another word.

```ts
import type { Context } from "@onetype/stack-api-kit";

import type { <Type> } from "./schemas/<Type>";
import type { <Plugin>Services } from "./types/<Plugin>Services";

const services = (ctx: Context) => ctx.use<<Plugin>Services>("<plugin>").<service>;

export const <Plugin> = {
    get: (ctx: Context, id: string): Promise<Type> =>
    {
        return services(ctx).get(id);
    },
};

export type { <Type> } from "./schemas/<Type>";
```
