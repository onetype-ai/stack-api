# plugin.ts

The contract: everything crossing the boundary, named here or absent. The
kernel refuses to start when a declaration is missing.

`<Plugin>` is the folder capitalised, `<service>` the service it holds. `over`
takes the database and services types, then the name and the contract: the
types are what `ctx.db` and `ctx.services` are inside every handler.

Required: `version`, `describe`.

```ts
export default definePlugin.over<<Plugin>Db, <Plugin>Services>()("<plugin>", {
    version: "1.0.0",
    describe: "<describe>",

    dependsOn: ["<other>"],
    config: <Plugin>Config.schema,

    tables: { <plugin> },
    migrations: "./src/plugins/<plugin>/migrations",
    scope: { describe: "<describe>", claim: "<claim>", tables: { <table>: "<field>" } },

    permissions: { "<plugin>.read": { describe: "<describe>" } },
    grantsSupported: ["<plugin>.read"],
    identifies: (ctx, request) => ({ id: "<identity>", claims: {} }),
    grants: (ctx, identity) => ["<plugin>.read"],

    services: (ctx) => ({ <service>: new <Plugin>(ctx) }),
    routes: [...<plugin>Routes],
    allowedHosts: ["https://<host>"],

    emits: { "<plugin>.<value>.<event>": { describe, schema } },
    channels: { "<plugin>.<value>": { describe, schema, reach: "scope" } },
    hooks: { "<plugin>.<value>.before-<action>": { describe, schema } },
    listens: { "<other>.<value>.<event>": defineListener(schema, { describe, handle }) },
    participates: { "<other>.<value>.before-<action>": defineParticipant(schema, { describe, handle }) },
    commands: { "<plugin>.<action>": defineCommand({ describe, schema, requires, run }) },

    setup: (ctx) => ctx.log.info("<plugin> ready", {}),
    teardown: (ctx) => ctx.log.info("<plugin> stopped"),
});
```
