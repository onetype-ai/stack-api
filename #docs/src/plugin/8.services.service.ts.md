# services/

Built once a request, so it holds `ctx` and never a caller of its own. A read
narrows with `scoped`, a write stamps with `stamped`, and every write goes
through `tx` or `write`. A read without `scoped` still compiles, and answers
every tenant's rows: verify names it `[unscoped]`. Write `scoped` in the
method that queries, and `stamped` inside the `values({ ... })` it stamps:
that is where verify looks.

```ts
export class Items
{
    readonly #ctx: ItemsContext;

    constructor(ctx: ItemsContext)
    {
        this.#ctx = ctx;
    }

    async get(id: string): Promise<Item>
    {
        const where = this.#ctx.scoped<SQL>("items");

        const [row] = await this.#ctx.db.select().from(items).where(and(where, eq(items.id, id)));

        if (row === undefined)
        {
            throw new Refusal(404, "NOT_FOUND", "No item by that id.");
        }

        return this.#toItem(row);
    }

    async create(title: string): Promise<Item>
    {
        return this.#ctx.tx(async (inside) =>
        {
            const [row] = await inside.db.insert(items).values({ id: crypto.randomUUID(), title: Item.title(title), createdAt: inside.now(), ...inside.stamped("items") } as Row).returning();

            if (row === undefined)
            {
                throw new Error("items: an insert answered no row, so the item cannot be announced.");
            }

            inside.events.emit("items.item.created", { id: row.id, tenantId: row.tenantId, title: row.title });

            return this.#toItem(row);
        });
    }

    #toItem(row: Row): Item
    {
        return { id: row.id, title: row.title, createdAt: row.createdAt };
    }
}
```

Inside `tx`, use `inside`, never the outer `ctx`: the outer one writes
outside the transaction just opened.
